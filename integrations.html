# checkmate_guard.py — boot-time identity guard
from checkmate_guard import Guard

guard = Guard(endpoint="https://checkmateee.vercel.app/api/sync")

@app.before_request
def verify_identity():
    decision = guard.verify(actor=current_identity, asset="")
    # DENY before any .env variable or connection string touches memory
    if decision.status != "ok":
        raise SystemExit(1)
checkmate-guard.js · imported in server.js 
// checkmate-guard.js — verify before secrets load
const { Guard } = require("./checkmate-guard");

const guard = new Guard({
  endpoint: "https://checkmateee.vercel.app/api/sync"
});

app.use(async (req, res, next) => {
  const decision = await guard.verify({
    actor: req.user, asset: ""
  });
  // revoked identity → terminate before secrets reach process memory
  if (decision.decision !== "ALLOW") process.exit(1);
  next();
});
